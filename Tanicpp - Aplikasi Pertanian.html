Kode Google Apps Script (`Code.gs`)

Salin kode di bawah ini ke editor Google Apps Script (`script.google.com`) untuk mendeploy Web App ini dengan database Google Sheets!

// ==========================================
// TANICPP - GOOGLE APPS SCRIPT BACKEND CODE
// File: Code.gs
// ==========================================

function doGet() {
  return HtmlService.createHtmlOutputFromFile('index')
    .setTitle('Tanicpp - Aplikasi Pertanian Smart Farming')
    .setXframeOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1');
}

// Fungsi Simpan Data Lahan / HST ke Google Sheets
function simpanDataLahan(data) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Lahan_HST') || ss.insertSheet('Lahan_HST');
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['ID', 'Nama Lahan', 'Komoditas', 'Tanggal Tanam', 'Jumlah Benih', 'Jenis Lahan']);
  }
  sheet.appendRow([data.id, data.nama, data.komoditas, data.tanggalTanam, data.jumlahBenih, data.jenisLahan]);
  return { status: 'success' };
}

// Fungsi Simpan Data Jadwal Perawatan ke Google Sheets
function simpanDataJadwal(data) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Jadwal_Perawatan') || ss.insertSheet('Jadwal_Perawatan');
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['ID', 'Lahan ID', 'Kategori', 'Nama Kegiatan', 'Tanggal', 'Jam', 'Interval', 'Status']);
  }
  sheet.appendRow([data.id, data.lahanId, data.kategori, data.nama, data.tanggalJadwal, data.jam, data.interval, data.status]);
  return { status: 'success' };
}

// Fungsi Simpan Modal / Biaya ke Google Sheets
function simpanDataModal(data) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Modal_Pengeluaran') || ss.insertSheet('Modal_Pengeluaran');
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['ID', 'Lahan ID', 'Kategori', 'Nama Item', 'Jumlah', 'Satuan', 'Total Harga', 'Tanggal']);
  }
  sheet.appendRow([data.id, data.lahanId, data.kategori, data.namaItem, data.jumlah, data.satuan, data.harga, data.tanggal]);
  return { status: 'success' };
}

// Fungsi Simpan Log Penggunaan ke Google Sheets
function simpanDataPenggunaan(data) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Log_Penggunaan') || ss.insertSheet('Log_Penggunaan');
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['ID', 'Lahan ID', 'Metode', 'Nama Bahan', 'Dosis', 'Volume', 'Target Hama', 'Tanggal']);
  }
  sheet.appendRow([data.id, data.lahanId, data.metode, data.namaBahan, data.dosis, data.volume, data.targetHama, data.tanggal]);
  return { status: 'success' };
}

Cara Deploy di Google Apps Script:

  1. Buka script.google.com dan buat proyek baru bernama Tanicpp.
  2. Tempelkan kode di atas pada file Code.gs.
  3. Buat file HTML baru bernama index.html dan tempelkan seluruh kode file ini.
  4. Klik Deploy > New deployment > Select type > Web app.
  5. Atur Who has access menjadi Anyone lalu klik Deploy. Selesai!
